to start of you need to run the following sql script in your database with privileged access.

```sql
BEGIN;

-- ==========================================
-- 0. EXTENSION & VERSION CHECK
-- ==========================================
DO $$
DECLARE
    v_installed_version TEXT;
BEGIN
    -- Ensure pg_cron extension exists
    IF NOT EXISTS (SELECT 1 FROM pg_extension WHERE extname = 'pg_cron') THEN
        RAISE EXCEPTION 'Prerequisite Failed: "pg_cron" extension is not installed.';
    END IF;

    -- Fetch installed version
    SELECT extversion INTO v_installed_version
    FROM pg_extension
    WHERE extname = 'pg_cron';

    -- Enforce minimum version 1.6.4
    IF v_installed_version < '1.6.4' THEN
        RAISE EXCEPTION 'Prerequisite Failed: "pg_cron" version must be >= 1.6.4 (Currently installed: %). Aborting installation.', v_installed_version;
    END IF;
END $$;


-- ==========================================
-- 1. SCHEMA & PERMISSIONS SETUP
-- ==========================================
CREATE SCHEMA IF NOT EXISTS pulsar;

DO $$ 
BEGIN
    IF NOT EXISTS (SELECT 1 FROM pg_type t JOIN pg_namespace n ON n.oid = t.typnamespace WHERE t.typname = 'host' AND n.nspname = 'pulsar') THEN
        CREATE TYPE pulsar.HOST AS ENUM (
            'PULSAR_STANDALONE',
            'NEBULA_CLUSTER'
        );
    END IF;
END $$;

-- Grant schema access
GRANT USAGE ON SCHEMA pulsar TO postgres, service_role, authenticated, anon;

-- Grant default privileges for current and future objects
ALTER DEFAULT PRIVILEGES IN SCHEMA pulsar GRANT ALL ON TABLES TO postgres, service_role, authenticated, anon;
ALTER DEFAULT PRIVILEGES IN SCHEMA pulsar GRANT ALL ON SEQUENCES TO postgres, service_role, authenticated, anon;

-- Set default search path for standard administrative user
ALTER USER postgres SET search_path TO pulsar, public;


-- ==========================================
-- 2. TABLES & INDEXES
-- ==========================================

-- 2.1 Restart Table
CREATE TABLE IF NOT EXISTS pulsar.pulsar_restart (
    id          BIGSERIAL PRIMARY KEY,
    queue_name TEXT NOT NULL UNIQUE,
    type        TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW() NOT NULL,
    owner       TEXT DEFAULT 'engine' NOT NULL
);


-- 2.2 Health Check Table
CREATE TABLE IF NOT EXISTS pulsar.health_check (
    id                        INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
    last_health_checked_at  TIMESTAMPTZ
);

CREATE TABLE IF NOT EXISTS pulsar.queue_locks (
    queue_name        TEXT PRIMARY KEY,
    locked_until     TIMESTAMPTZ NOT NULL,
    fencing_token    INTEGER GENERATED BY DEFAULT AS IDENTITY,
    last_valid_token INTEGER DEFAULT 0
);

-- 2.3 Unlogged Queue Cacbe Table
CREATE TABLE IF NOT EXISTS pulsar.queue_cache (
    key         VARCHAR,
    field       VARCHAR,
    value       VARCHAR NOT NULL,
    expires_at BIGINT NOT NULL,
    PRIMARY KEY (key, field)
);

CREATE TABLE IF NOT EXISTS pulsar.pool_queue_entries (
    id         INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
    queue      TEXT NOT NULL,
    created_at TIMESTAMPTZ,
    payload    JSONB NOT NULL
);

CREATE TABLE IF NOT EXISTS pulsar.meta (
    id             INTEGER PRIMARY KEY GENERATED ALWAYS AS IDENTITY,
    host           pulsar.HOST NOT NULL,
    version        VARCHAR(50) NOT NULL,
    url            TEXT NOT NULL,
    hash           TEXT,
    configurations JSONB DEFAULT '{}'::jsonb,
    is_active      BOOLEAN DEFAULT TRUE,
    created_at     TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    updated_at     TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    instance_id    TEXT NOT NULL UNIQUE
);

CREATE INDEX IF NOT EXISTS idx_host_lookup ON pulsar.meta (host, is_active);


-- ==========================================
-- 2.4 ENABLE ROW LEVEL SECURITY (RLS)
-- ==========================================
ALTER TABLE pulsar.pulsar_restart ENABLE ROW LEVEL SECURITY;
ALTER TABLE pulsar.health_check ENABLE ROW LEVEL SECURITY;
ALTER TABLE pulsar.queue_locks ENABLE ROW LEVEL SECURITY;
ALTER TABLE pulsar.queue_cache ENABLE ROW LEVEL SECURITY;
ALTER TABLE pulsar.pool_queue_entries ENABLE ROW LEVEL SECURITY;
ALTER TABLE pulsar.meta ENABLE ROW LEVEL SECURITY;

-- Create barebone permissive policies so table access is not blocked
CREATE POLICY "Enable all access for pulsar_restart" ON pulsar.pulsar_restart FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Enable all access for health_check" ON pulsar.health_check FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Enable all access for queue_locks" ON pulsar.queue_locks FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Enable all access for queue_cache" ON pulsar.queue_cache FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Enable all access for pool_queue_entries" ON pulsar.pool_queue_entries FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Enable all access for meta" ON pulsar.meta FOR ALL USING (true) WITH CHECK (true);


-- Grant permissions on newly created tables/sequences in batch
GRANT ALL PRIVILEGES ON ALL TABLES IN SCHEMA pulsar TO postgres, service_role, authenticated, anon;
GRANT ALL PRIVILEGES ON ALL SEQUENCES IN SCHEMA pulsar TO postgres, service_role, authenticated, anon;


-- ==========================================
-- 3. HTTP DELAYED SCHEDULER FUNCTIONS
-- ==========================================

-- Clean up existing function signatures if re-running
DROP FUNCTION IF EXISTS pulsar.fire_delayed_http(text, timestamptz, text, text, jsonb, jsonb, int) CASCADE;
DROP FUNCTION IF EXISTS pulsar.schedule_http_request(text, numeric, text, jsonb, jsonb, int) CASCADE;

-- Execution worker called inside dynamic pg_cron jobs
CREATE OR REPLACE FUNCTION pulsar.fire_delayed_http (
    p_job        TEXT,
    p_run_at     TIMESTAMPTZ,
    p_method     TEXT,
    p_url        TEXT,
    p_headers    JSONB,
    p_body       JSONB,
    p_timeout_ms INT
) RETURNS VOID LANGUAGE plpgsql SECURITY DEFINER
SET search_path = '' AS $$
BEGIN
    IF clock_timestamp() < p_run_at THEN
        RETURN;  -- Job is not due yet, poll next tick
    END IF;

    BEGIN
        IF upper(p_method) = 'GET' THEN
            PERFORM net.http_get(
                url                := p_url, 
                headers            := p_headers,
                timeout_milliseconds := p_timeout_ms
            );
        ELSIF upper(p_method) = 'DELETE' THEN
            PERFORM net.http_delete(
                url                := p_url, 
                headers            := p_headers,
                timeout_milliseconds := p_timeout_ms
            );
        ELSE
            PERFORM net.http_post(
                url                := p_url, 
                body               := p_body, 
                headers            := p_headers,
                timeout_milliseconds := p_timeout_ms
            );
        END IF;
    EXCEPTION WHEN OTHERS THEN
        RAISE WARNING 'Delayed HTTP job % failed: %', p_job, SQLERRM;
    END;

    -- Unschedule after payload dispatch
    PERFORM cron.unschedule(p_job);
END;
$$;


-- Entrypoint function to schedule HTTP tasks
CREATE OR REPLACE FUNCTION pulsar.schedule_http_request (
    p_url             TEXT,
    p_delay_seconds NUMERIC DEFAULT 0,
    p_method          TEXT DEFAULT 'POST',
    p_body            JSONB DEFAULT '{}',
    p_headers         JSONB DEFAULT '{"Content-Type": "application/json"}',
    p_timeout_ms    INT DEFAULT 5000
) RETURNS TEXT LANGUAGE plpgsql SECURITY DEFINER
SET search_path = '' AS $$
DECLARE
    v_job    TEXT        := 'delayed-http-' || gen_random_uuid()::text;
    v_run_at TIMESTAMPTZ := clock_timestamp() + make_interval(secs => p_delay_seconds);
BEGIN
    PERFORM cron.schedule(
        v_job,
        '1 seconds',
        format(
            'SELECT pulsar.fire_delayed_http(%L, %L::timestamptz, %L, %L, %L::jsonb, %L::jsonb, %s)',
            v_job, v_run_at, p_method, p_url, p_headers, p_body, p_timeout_ms
        )
    );
    RETURN v_job;
END;
$$;

-- Secure HTTP functions (Least privilege model)
REVOKE EXECUTE ON FUNCTION pulsar.schedule_http_request(text, numeric, text, jsonb, jsonb, int) FROM PUBLIC, anon, authenticated;
REVOKE EXECUTE ON FUNCTION pulsar.fire_delayed_http(text, timestamptz, text, text, jsonb, jsonb, int) FROM PUBLIC, anon, authenticated;

GRANT EXECUTE ON FUNCTION pulsar.schedule_http_request(text, numeric, text, jsonb, jsonb, int) TO service_role;


DROP FUNCTION IF EXISTS pulsar.handle_edp_queue_dispatch();

CREATE OR REPLACE FUNCTION pulsar.handle_edp_queue_dispatch() RETURNS trigger AS $$
DECLARE
    v_queue_name text := COALESCE(TG_ARGV[0], 'q_1');
    v_instance_id text := COALESCE(TG_ARGV[1], 'unknown');
    v_target_url text;
    v_trigger_strategy text;
    v_lock_key bigint := hashtext(v_queue_name);
    v_acquired boolean;
    v_fencing_token integer;
    v_headers jsonb;
    v_payload jsonb;
    v_channel_name text := 'channel_pulsar_event_driven_queue';
BEGIN
    v_acquired := pg_try_advisory_xact_lock(v_lock_key);

    IF NOT v_acquired THEN
        RETURN NULL;
    END IF;

    SELECT url, configurations->>'event_trigger_strategy' 
    INTO v_target_url, v_trigger_strategy
    FROM pulsar.meta
    WHERE is_active = true
        AND instance_id = v_instance_id
        AND EXISTS (
          SELECT 1 
          FROM jsonb_array_elements(configurations->'queues') AS q
          WHERE q->>'queue' = v_queue_name 
            AND q->>'pooler' = 'event-driven-pooler'
      )
    LIMIT 1;

    IF v_trigger_strategy IS NULL THEN
        RETURN NULL;
    END IF;

    INSERT INTO pulsar.queue_locks (queue_name, locked_until)
    VALUES (v_queue_name, clock_timestamp() + INTERVAL '30 seconds')
    ON CONFLICT (queue_name) DO UPDATE 
        SET locked_until = clock_timestamp() + INTERVAL '30 seconds',
            fencing_token = nextval('pulsar.unlogged_queue_locks_fencing_token_seq')
        WHERE pulsar.queue_locks.locked_until <= clock_timestamp()
    RETURNING fencing_token INTO v_fencing_token;

    IF v_fencing_token IS NULL THEN
        RETURN NULL;
    END IF;

    v_payload := jsonb_build_object(
        'queue_name', v_queue_name,
        'fencing_token', v_fencing_token,
        'execution_at', to_char(clock_timestamp() AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.MS"Z"')
    );

    CASE UPPER(v_trigger_strategy)
        WHEN 'WEBHOOK' THEN
            IF v_target_url IS NULL OR v_target_url = 'NOT_REQUIRED' THEN
                RETURN NULL;
            END IF;

            v_headers := jsonb_build_object(
                'Content-Type', 'application/json',
                'x-fencing-token', v_fencing_token,
                'x-last-execution-id', NULL,
                'x-last-execution-at', to_char(clock_timestamp() AT TIME ZONE 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.MS"Z"'),
                'x-execution-owner', 'trigger',
                'x-restart', 'false',
                'x-pooler', 'event-driven-pooler',
                'x-target-queue', v_queue_name
            );

            PERFORM net.http_post(
                url := v_target_url,
                body := jsonb_build_object('queue_name', v_queue_name),
                headers := v_headers
            );

        WHEN 'POSTGRES_NOTIFY' THEN
            PERFORM pg_notify(v_channel_name, v_payload::text);

        WHEN 'POLLING' THEN
            INSERT INTO pulsar.pool_queue_entries (queue, payload, created_at)
            VALUES (v_queue_name, v_payload, clock_timestamp());

        ELSE
            NULL;
    END CASE;

    RETURN NULL;
END;
$$ LANGUAGE plpgsql;

-- 4. PULSAR WATCHDOG

DROP FUNCTION IF EXISTS pulsar.watchdog_engine_heartbeat() CASCADE;

CREATE OR REPLACE FUNCTION pulsar.watchdog_engine_heartbeat() 
RETURNS VOID LANGUAGE plpgsql AS $$
DECLARE
    v_last_ping_at TIMESTAMPTZ;
BEGIN
    SELECT last_health_checked_at INTO v_last_ping_at
    FROM pulsar.health_check
    WHERE id = 1;

    IF v_last_ping_at IS NULL OR v_last_ping_at < (NOW() - INTERVAL '1 minute') THEN
        PERFORM net.http_post(
            url     := 'http://host.docker.internal:9999/process-job',
            body    := '{"reason": "engine_unresponsive_wake_up"}'::jsonb,
            headers := '{"Content-Type": "application/json"}'::jsonb
        );
    END IF;
END;
$$;

DROP TRIGGER IF EXISTS trg_set_meta_hash ON pulsar.meta;

CREATE OR REPLACE FUNCTION pulsar.generate_sha256(input_string TEXT)
RETURNS TEXT AS $$
BEGIN
    RETURN encode(sha256(input_string::bytea), 'hex');
END;
$$ LANGUAGE plpgsql IMMUTABLE;


-- Create the trigger function that automatically computes and sets the hash from jsonb
CREATE OR REPLACE FUNCTION pulsar.set_meta_hash_trigger()
RETURNS TRIGGER AS $$
BEGIN
    NEW.hash := pulsar.generate_sha256(NEW.configurations::TEXT);
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_set_meta_hash
    BEFORE INSERT OR UPDATE OF configurations ON pulsar.meta
    FOR EACH ROW
    EXECUTE FUNCTION pulsar.set_meta_hash_trigger();


-- ==========================================
-- 5. MILLISECOND TIME
-- ==========================================

CREATE OR REPLACE FUNCTION pulsar.now_ms() RETURNS bigint AS $$
    SELECT (extract(epoch FROM clock_timestamp()) * 1000)::bigint
$$ LANGUAGE sql VOLATILE;


-- ==========================================
-- 6. PGMQ EXTENSION INTEGRATION
-- ==========================================

DROP FUNCTION IF EXISTS pulsar.get_single_queue_metrics(TEXT) CASCADE;

CREATE OR REPLACE FUNCTION pulsar.get_single_queue_metrics(target_queue TEXT) 
RETURNS TABLE (
    queue_name      TEXT,
    ready_jobs      BIGINT,
    delayed_jobs    BIGINT,
    total_jobs      BIGINT,
    next_visible_at  TIMESTAMPTZ
) LANGUAGE plpgsql AS $$
BEGIN
    queue_name := target_queue;

    RETURN QUERY EXECUTE format(
        'SELECT 
            %L::TEXT,
            COALESCE(COUNT(*) FILTER (WHERE vt <= clock_timestamp()), 0),
            COALESCE(COUNT(*) FILTER (WHERE vt > clock_timestamp()), 0),
            COALESCE(COUNT(*), 0),
            MIN(vt) FILTER (WHERE vt > clock_timestamp())
         FROM pgmq.%I', 
        target_queue,
        'q_' || target_queue
    );
END;
$$;

COMMIT;
```

```sql
-- Old signature took p_now from the app; drop it so nobody can call it by accident.
DROP FUNCTION IF EXISTS register_workers(VARCHAR, VARCHAR, INT, INT, BIGINT, BIGINT, VARCHAR[]);
DROP FUNCTION IF EXISTS register_workers(VARCHAR, VARCHAR, INT, INT, BIGINT, VARCHAR[]);

CREATE OR REPLACE FUNCTION register_workers (
    p_queue_key VARCHAR,
    p_global_key VARCHAR,
    p_local_concurrency INT,
    p_global_concurrency INT,
    p_ttl_ms BIGINT,
    p_candidate_ids VARCHAR[]
) RETURNS TABLE (granted_worker_id VARCHAR) AS $$
DECLARE
    v_now BIGINT;
    v_expire_at BIGINT;
    v_local_active INT;
    v_global_active INT;
    v_available_slots INT;
    v_candidate_count INT;
    v_to_grant INT;
BEGIN
    -- Serialize calls touching this queue and the global pool (consistent order => no deadlocks)
    IF hashtext(p_queue_key) < hashtext(p_global_key) THEN
        PERFORM pg_advisory_xact_lock(hashtext(p_queue_key));
        PERFORM pg_advisory_xact_lock(hashtext(p_global_key));
    ELSE
        PERFORM pg_advisory_xact_lock(hashtext(p_global_key));
        PERFORM pg_advisory_xact_lock(hashtext(p_queue_key));
    END IF;

    -- Read the clock AFTER taking the locks so waiting on them can't make "now" stale.
    v_now := pulsar.now_ms();
    v_expire_at := v_now + p_ttl_ms;

    -- 1. Purge expired workers (queue + global)
    DELETE FROM pulsar.queue_cache
    WHERE (key = p_queue_key OR key = p_global_key)
      AND expires_at <= v_now;

    -- 2. Active counts
    SELECT COUNT(*) INTO v_local_active
    FROM pulsar.queue_cache
    WHERE key = p_queue_key AND expires_at > v_now;

    SELECT COUNT(*) INTO v_global_active
    FROM pulsar.queue_cache
    WHERE key = p_global_key AND expires_at > v_now;

    -- 3. Global check, then local check (same order as the Lua script)
    IF p_global_concurrency - v_global_active <= 0 THEN
        RETURN;
    END IF;

    IF p_local_concurrency - v_local_active <= 0 THEN
        RETURN;
    END IF;

    -- 4. Atomic capacity = MIN(local, global)
    v_available_slots := LEAST(
        p_global_concurrency - v_global_active,
        p_local_concurrency - v_local_active
    );

    v_candidate_count := COALESCE(cardinality(p_candidate_ids), 0);
    v_to_grant := LEAST(v_available_slots, v_candidate_count);

    IF v_to_grant <= 0 THEN
        RETURN;
    END IF;

    -- 5. Grant first N candidates (input order preserved, duplicates collapsed)
    RETURN QUERY
    WITH candidate_slice AS (
        SELECT t.worker_id, MIN(t.ord) AS ord
        FROM unnest(p_candidate_ids[1:v_to_grant]) WITH ORDINALITY AS t(worker_id, ord)
        GROUP BY t.worker_id
    ),
    inserted_local AS (
        INSERT INTO pulsar.queue_cache (key, field, value, expires_at)
        SELECT p_queue_key, cs.worker_id, 'active', v_expire_at
        FROM candidate_slice cs
        ON CONFLICT (key, field)
        DO UPDATE SET expires_at = EXCLUDED.expires_at
        RETURNING field
    ),
    inserted_global AS (
        INSERT INTO pulsar.queue_cache (key, field, value, expires_at)
        SELECT p_global_key, p_queue_key || ':' || cs.worker_id, 'active', v_expire_at
        FROM candidate_slice cs
        ON CONFLICT (key, field)
        DO UPDATE SET expires_at = EXCLUDED.expires_at
        RETURNING field
    )
    SELECT cs.worker_id
    FROM candidate_slice cs
    ORDER BY cs.ord;
END;
$$ LANGUAGE plpgsql;
```
